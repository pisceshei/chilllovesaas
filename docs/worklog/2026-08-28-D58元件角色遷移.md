```python
# 硬編字型值與角色階消費的對帳 —— 🔴 附 canary
import re, io
SH = {'450', '550', '600', '650'}
tot_role = 0
for f, lab in (('app/assets/stylesheets/admin.css', 'admin.css'),
               ('docs/design/chilllove-admin-v2.html', '原型')):
    s = re.sub(r'/\*.*?\*/', '', io.open(f, encoding='utf-8').read(), flags=re.S)
    fw = re.findall(r'font-weight:\s*(\d+)', s)
    fs = re.findall(r'font-size:\s*([\d.]+px)', s)
    lh = re.findall(r'line-height:\s*([\d.]+)\s*[;}]', s)
    role = len(re.findall(r'var\(--type-', s))
    tot_role += role
    print(lab, '硬編fw', len(fw), '域外', sum(1 for x in fw if x not in SH),
          '硬編fs', len(fs), '無單位lh', len(lh), '角色階', role)
assert tot_role > 0, 'canary 失敗：一個角色階消費都沒掃到，上面的數字不可信'
# 期望：admin.css 硬編fw 5 域外 3 / 硬編fs 8 / 無單位lh 4 / 角色階 116
#      原型   硬編fw 93 域外 32 / 硬編fs 11 / 無單位lh 26 / 角色階 16
```
