#### PR #66 ⚪ 動態集合對帳

下式不手抄 comment ID。白色項目的正式 grammar 是：行首空白後可有一個 CommonMark
無序／編號 marker，接著可有零至兩個 `*`，再立即接 `⚪-<整數>`；標題接受一至六級 heading
或粗體形式。`candidate` 不依標題抽取，`observed` 只在白色區段抽取；兩者逐 comment 必須相等，
再與 `91` §3 的 `registered` pair 集合相等。任一區段漏辨、漏登或序號不符都以非零終止。

```powershell
$ErrorActionPreference = 'Stop'
$pages = gh api --paginate --slurp 'repos/pisceshei/chilllovesaas/issues/66/comments?per_page=100' |
  ConvertFrom-Json
if ($LASTEXITCODE -ne 0) { throw 'GitHub comments unavailable' }
$itemPattern = '^\s*(?:[-*+]\s+|\d+[.)]\s+)?\*{0,2}⚪-(\d+)'
$headerPattern = '^\s*(?:#{1,6}\s+|\*{2})⚪ 範圍外觀察'
function Get-WhitePairs([object[]]$Comments) {
  $allObserved = @(
    foreach ($comment in $Comments) {
    if ($comment.user.login -ne 'claude[bot]') { continue }
    $candidate = @()
    $commentObserved = @()
    $inScope = $false
    foreach ($line in @($comment.body -split "`n")) {
      if ($line -match $itemPattern) {
        $candidate += '{0}:{1}' -f $comment.id, $Matches[1]
      }
      if ($line -match $headerPattern) {
        $inScope = $true
        continue
      }
      if (-not $inScope) { continue }
      if ($line -match $itemPattern) {
        $commentObserved += '{0}:{1}' -f $comment.id, $Matches[1]
      } elseif ($line -match '^\s*#{1,6}\s+' -or $line -match '^\s*\*{2}(?!⚪-)') {
        $inScope = $false
      }
    }
    $candidate = @($candidate | Sort-Object -Unique)
    $commentObserved = @($commentObserved | Sort-Object -Unique)
    if ($candidate.Count -gt 0 -and $commentObserved.Count -eq 0) {
      throw "white ledger zero-scan: comment $($comment.id)"
    }
    if (@(Compare-Object $candidate $commentObserved).Count -ne 0) {
      throw "white ledger partial-scan: comment $($comment.id)"
    }
    $commentObserved
  }
  )
  @($allObserved | Sort-Object -Unique)
}
$comments = @($pages | ForEach-Object { $_ })
$observed = @(Get-WhitePairs $comments)
$pit = Get-Content -LiteralPath 'docs/specs/91-pit-register.md' -Raw -Encoding UTF8
$section = [regex]::Match($pit, '(?ms)^## §3 .*?(?=^## 附錄 A)')
if (-not $section.Success) { throw 'pit register section 3 missing' }
$registered = @(
  foreach ($match in [regex]::Matches(
    $section.Value,
    'PR #66 Claude issue comment\s+`(\d+)` ⚪(\d+)'
  )) {
    '{0}:{1}' -f $match.Groups[1].Value, $match.Groups[2].Value
  }
) | Sort-Object -Unique
$delta = @(Compare-Object $observed $registered)
if ($delta.Count -ne 0) {
  $delta | Format-Table -AutoSize | Out-String | Write-Error
  throw 'white ledger mismatch'
}
'WHITE_LEDGER_MATCH={0}' -f $observed.Count

$variant = [pscustomobject]@{
  id = 999000001
  user = [pscustomobject]@{ login = 'claude[bot]' }
  body = "### ⚪ 範圍外觀察`n1. ⚪-1 合法格式變體"
}
$variantPairs = @(Get-WhitePairs @($variant))
if ($variantPairs -ne '999000001:1') { throw 'white ledger variant rejected' }
'WHITE_LEDGER_VARIANT_MATCH=1'

$missingHeader = [pscustomobject]@{
  id = 999000002
  user = [pscustomobject]@{ login = 'claude[bot]' }
  body = '- ⚪-1 缺少白色區段標題'
}
$canaryRed = $false
try { $null = @(Get-WhitePairs @($missingHeader)) } catch { $canaryRed = $true }
if (-not $canaryRed) { throw 'white ledger zero-scan canary failed' }
'WHITE_LEDGER_CANARY_RED=1'

$partial = [pscustomobject]@{
  id = 999000003
  user = [pscustomobject]@{ login = 'claude[bot]' }
  body = "## ⚪ 範圍外觀察`n- **⚪-1** 區段內`n## ✅ 已符合`n- ⚪-2 區段外候選"
}
$partialRed = $false
try { $null = @(Get-WhitePairs @($partial)) } catch { $partialRed = $true }
if (-not $partialRed) { throw 'white ledger partial-scan canary failed' }
'WHITE_LEDGER_PARTIAL_RED=1'
```
