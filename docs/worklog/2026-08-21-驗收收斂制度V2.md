#### PR #66 ⚪ 動態集合對帳

下式不手抄 comment ID。白色項目的正式 grammar 是：行首空白後可有一個 CommonMark
無序／編號 marker，接著可有零至兩個 `*`，再立即接 `⚪-<整數>`；標題接受一至六級 heading
或粗體形式。`candidate` 不依標題抽取，`observed` 只在白色區段抽取；兩者逐 comment 必須相等，
再與 `91` §3 的 `registered` pair 加上 PR body exact-head `DEFERRED_WHITE` pair 聯集合相等。
deferred 只允許當前 head、不得重複且來源 pair 必須真實存在；任一區段漏辨、漏登、錯 head、
重複或序號不符都以非零終止。

[Console]::OutputEncoding = [System.Text.Encoding]::UTF8

)
if ($reviewersMatches.Count -ne 1) { throw 'workflow reviewer allowlist missing or ambiguous' }

$headerPattern = '^\s*(?:#{1,6}\s+|\*{2})⚪ 範圍外觀察'
function Get-WhitePairs([object[]]$Comments) {
  $allObserved = @(
    foreach ($comment in $Comments) {
    if ($comment.user.login -notin $acceptedReviewers) { continue }
    $candidate = @()
    $commentObserved = @()
    $inScope = $false
    foreach ($line in @($comment.body -split "`n")) {
      if ($line -match $itemPattern) {
        $candidate += '{0}:{1}' -f $comment.id, $Matches[1]
      }
      if ($line -match $headerPattern) {
        $inScope = $true
        continue
      }
      if (-not $inScope) { continue }
      if ($line -match $itemPattern) {
        $commentObserved += '{0}:{1}' -f $comment.id, $Matches[1]
      } elseif ($line -match '^\s*#{1,6}\s+' -or $line -match '^\s*\*{2}(?!⚪-)') {
        $inScope = $false
      }
    }
    # 🔴 **同一則判詞內重複的 ⚪ 編號必須先擋下**（2026-08-21 修正，來源＝Codex inline
    # `3831890283`）：若一則判詞把兩條不同觀察都編成 `⚪-1`，`Sort-Object -Unique` 會把
    # candidate 與 in-section 兩側都摺成同一個 `comment:1`，後面的比對照樣通過 ⇒ 只登記一條
    # 也算數，另一條靜默蒸發。編號重複本身即判詞格式錯誤，fail-closed。
    # 只驗 in-section 側：candidate 掃全文，判詞散文中再次提及某編號屬正常，不該轉紅。
    foreach ($pair in @($commentObserved | Sort-Object -Unique)) {
      if (@($commentObserved | Where-Object { $_ -eq $pair }).Count -ne 1) {
        throw "duplicate white item identifier in comment $($comment.id): $pair"
      }
    }
    $candidate = @($candidate | Sort-Object -Unique)
    $commentObserved = @($commentObserved | Sort-Object -Unique)
    if ($candidate.Count -gt 0 -and $commentObserved.Count -eq 0) {
      throw "white ledger zero-scan: comment $($comment.id)"
    }
    if (@(Compare-Object $candidate $commentObserved).Count -ne 0) {
      throw "white ledger partial-scan: comment $($comment.id)"
    }
    $commentObserved
  }
  )
  @($allObserved | Sort-Object -Unique)
}

  # 🔴 **偏好必須先降級，否則具名判準不承重**（2026-08-22 實跑發現）：在
  # `$ErrorActionPreference = 'Stop'` 下，PS 5.1 會把原生 git 的 stderr 包成 ErrorRecord 並
  # **先於**下面的退出碼檢查終止本函式 ⇒ 紅燈訊息變成 git 原文（`fatal: Invalid revision
  # range ...`），我們自己的具名訊息永遠不會出現。方向雖仍是 fail-closed，但判準是 PowerShell
  # 的偏好在擋、不是本函式在擋，突變因此證不出本函式承重。
  # 這裡把偏好降為 Continue（**函式作用域**，離開自動還原），改以退出碼判定；`throw` 不受影響。
  $ErrorActionPreference = 'Continue'
  $heads = @(git log --format=%H "$Base..HEAD" 2>$null)
  $gitExit = $LASTEXITCODE
  if ($gitExit -ne 0) { throw "ledger ancestry git log failed: exit=$gitExit base=$Base" }
  if ($heads.Count -eq 0) { throw "ledger ancestry git log returned no commit for $Base..HEAD" }

}
# 🔴 **mode 必須由呼叫端明示宣告，不得由 heads 的形狀反推**（同來源）：「零重疊」既可能是
# squash 合併後，也可能是 ledger base 寫錯或歷史被改寫，**兩者結論相反**——前者該放行，
# 後者該轉紅。因此 mode 取自**可觀察事實**（PR 狀態），不是從祖先鏈推的。
function Invoke-AncestryGuard([string[]]$Registry, [string[]]$Heads, [string]$Mode) {
  if ($Mode -ne 'branch' -and $Mode -ne 'merged') {
    throw "ledger ancestry mode must be declared explicitly as branch|merged: got '$Mode'"
  }
  if (@($Registry).Count -eq 0) { throw 'ledger ancestry registry is empty' }
  if (@($Heads).Count -eq 0) { throw 'ledger ancestry received no heads' }
  $overlap = @($Heads | Where-Object { $Registry -contains $_ }).Count
  if ($overlap -eq 0) {
    if ($Mode -eq 'merged') { return 'LEDGER_ANCESTRY_NOT_APPLICABLE=1' }
    throw 'branch mode found no registry head on the ancestry: ledger base or registry is wrong'
  }
  foreach ($branchHead in @($Heads | Select-Object -Skip 1)) {
    if ($Registry -notcontains $branchHead) {
      throw "reviewed branch head missing from ledger registry: $branchHead"
    }
  }
  'LEDGER_ANCESTRY_CHECK=1'
}
