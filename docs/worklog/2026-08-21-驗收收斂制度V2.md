function Assert-FrozenLedgerCoverage([string]$Text, [string[]]$BodySet, [string[]]$WorklogSet, [string[]]$ExpectedHeads, [string[]]$AllKnownHeads) {
  $markerPattern = '(?m)^<!-- FROZEN_LEDGER_HEAD=([0-9a-f]{40}) -->$'
  $markerMatches = @([regex]::Matches($Text, $markerPattern))
  $markerHeads = @($markerMatches | ForEach-Object { $_.Groups[1].Value } | Sort-Object -Unique)
  if ($markerMatches.Count -ne $markerHeads.Count -or $markerHeads.Count -eq 0) {
    throw 'frozen ledger marker missing or duplicated'
  }
  # 期望集合由 git 導出，不由本檔標題自我導出 ⇒ 少寫一個 section 必然轉紅。
  # 兩個方向都檢查，但刻意分開：①每個「已受驗 head」都必須有 marker（承重方向）；
  # ②每個 marker 都必須是分支上的 commit 或明列的不可達 head（防止憑空造 marker）。
  # 尚未受驗的 tip 允許但不強制有 marker，故本式在 commit 前後都成立。
  foreach ($expected in $ExpectedHeads) {
    if ($expected -notin $markerHeads) {
      throw "frozen ledger section missing for reviewed head: $expected"
    }

      throw "frozen ledger marker does not correspond to any known head: $marker"

    [regex]::Matches($Text, '(?m)^### exact-head `([0-9a-f]{7})`') |
      ForEach-Object { $_.Groups[1].Value } |
      Sort-Object -Unique
  )
  foreach ($prefix in $exactPrefixes) {
    if (@($markerHeads | Where-Object { $_.StartsWith($prefix) }).Count -ne 1) {
      throw "exact-head section marker missing or duplicated: $prefix"
    }
  }
  if ($BodySet.Count -ne $markerHeads.Count -or $WorklogSet.Count -ne $markerHeads.Count) {
    throw 'frozen ledger tuple set mismatch'
  }
  foreach ($head in $markerHeads) {
    $headPattern = '^LEDGER_TUPLE head=' + [regex]::Escape($head) + ' '
